cmake_minimum_required(VERSION 3.21)
project(quickDoc LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "quickDoc requires Windows x64 and MSVC (Visual Studio 2022)")
endif()
set(PDFIUM_DIR "${CMAKE_SOURCE_DIR}/.deps/pdfium")
if(NOT EXISTS "${PDFIUM_DIR}/public/fpdfview.h" OR NOT EXISTS "${PDFIUM_DIR}/lib/pdfium.dll.lib" OR NOT EXISTS "${PDFIUM_DIR}/bin/pdfium.dll")
  message(FATAL_ERROR "PDFium missing. Run setup.bat first.")
endif()
option(EMBED_PDFIUM "Embed PDFium DLL in the executable for single-file distribution" OFF)
set(SOURCES main.cpp quickDoc.rc)
if(EMBED_PDFIUM)
  configure_file("${CMAKE_SOURCE_DIR}/pdfium.rc.in" "${CMAKE_BINARY_DIR}/pdfium.rc" @ONLY)
  list(APPEND SOURCES "${CMAKE_BINARY_DIR}/pdfium.rc")
endif()
add_executable(quickDoc WIN32 ${SOURCES})
target_compile_features(quickDoc PRIVATE cxx_std_20)
target_compile_options(quickDoc PRIVATE /W4 /EHsc /permissive- /utf-8)
set_property(TARGET quickDoc PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(quickDoc PRIVATE "${CMAKE_SOURCE_DIR}/.deps")
target_link_libraries(quickDoc PRIVATE "${PDFIUM_DIR}/lib/pdfium.dll.lib" d2d1 comctl32 comdlg32 shell32 ole32)
if(EMBED_PDFIUM)
  target_compile_definitions(quickDoc PRIVATE EMBED_PDFIUM=1)
  target_link_libraries(quickDoc PRIVATE delayimp)
  target_link_options(quickDoc PRIVATE "/DELAYLOAD:pdfium.dll")
else()
  add_custom_command(TARGET quickDoc POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${PDFIUM_DIR}/bin/pdfium.dll" "$<TARGET_FILE_DIR:quickDoc>/pdfium.dll")
endif()
